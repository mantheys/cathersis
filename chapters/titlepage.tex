% in the thesis.tex preamble. Unknown styles fall back to "default".

  {\input{chapters/titlepages/\titlepagestyle}}
  {\PackageWarning{titlepage}{Unknown title page style `\titlepagestyle'; using `default'}%
   \input{chapters/titlepages/default}}
