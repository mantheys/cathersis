% Minimal title page (style: minimal).
% Left-aligned on the text block, in the body fonts and the accent palette of
% references/style/main.tex: a short accent rule over the title, the author
% below it, and the degree information set small at the foot of the page.
\newcommand{\minimalmuted}{\color{black!60}}
% Directors from the comma-separated \thesisdirectors, joined by commas; the
% "Supervised by" line is omitted when the list is empty.
\ExplSyntaxOn
\NewDocumentCommand{\minimaldirection}{}{
  \clist_set:No \l_tmpa_clist {\thesisdirectors}
  \clist_if_empty:NF \l_tmpa_clist {
    {\minimalmuted Supervised~by}\quad \clist_use:Nn \l_tmpa_clist {,~} \par
  }
}
\ExplSyntaxOff
% Month and year of the build, e.g. "October 2026"
\newcommand{\minimaldate}{\ifcase\month\or January\or February\or March\or April\or May\or June\or
  July\or August\or September\or October\or November\or December\fi\ \number\year}

\begin{titlepage}
  \raggedright
  \setlength{\parskip}{0pt}

  {\small\minimalmuted\MakeUppercase{Doctoral Thesis}\par}
  \vspace*{0.2\textheight}

  {\color{accentOrange}\rule{2.5cm}{1.5pt}\par}
  \vspace{1.2em}

  {\fontsize{28}{34}\selectfont\bfseries\hyphenchar\font=-1 \thesistitle\par}
  \vspace{1em}
  {\Large\minimalmuted \thesissubtitle\par}

  \vspace{3.5em}
  {\Large \thesisauthor\par}

  \vfill

  {\small
    \begin{spacing}{1.3}
      Programme in \thesisprogramme\par
      \thesisuniversity\ \textperiodcentered\ \thesisinstitution\par
      \minimaldirection
      \vspace{0.6em}
      {\minimalmuted \minimaldate\par}
    \end{spacing}}
\end{titlepage}
