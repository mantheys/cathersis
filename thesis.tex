% Written to .aux on each pass so mappings are available when the TOC/minitoc is
% typeset (which happens before the chapter bodies where \pcstart is called).
\newcommand{\pcannmark}[3]{%  #1=id  #2=target  #3=hyperref-anchor
  \expandafter\gdef\csname pct:#1\endcsname{#2}%
  \expandafter\gdef\csname pcid@#3\endcsname{#1}%
}

% --- Annotating \contentsline for same-line page-count info in TOC/minitoc ---
%
% Strategy: wrap \tableofcontents and \minitoc to replace \contentsline with a
% version that looks up the pcid@ mapping (written to .aux by \pcstart) for the
% hyperref anchor in argument #4 and appends the annotation inline to the title.
% No buffering needed: pcid@ data is available immediately from the .aux macros.

\newcommand{\@pcann@annotated@cl}[4]{%
  \ifcsname pcid@#4\endcsname
    \edef\@pcann@id{\csname pcid@#4\endcsname}%
    \@orig@contentsline{#1}%
      {#2\texorpdfstring{%
         \ifpagecount
           \nobreakspace
           {\color{accentSkyBlue}\scriptsize\itshape
             [\csname pct:\@pcann@id\endcsname\,pp\enspace$|$\enspace
              \the\numexpr
                \getpagerefnumber{pce:\@pcann@id}%
                -\getpagerefnumber{pcs:\@pcann@id}\relax\,pp]}%
         \fi}{}}%
      {#3}{#4}%
  \else
    \@orig@contentsline{#1}{#2}{#3}{#4}%
  \fi
}

\let\@orig@tableofcontents\tableofcontents
\renewcommand{\tableofcontents}{%
  \let\@orig@contentsline\contentsline
  \let\contentsline\@pcann@annotated@cl
  \@orig@tableofcontents
  \let\contentsline\@orig@contentsline
}

\let\@orig@minitoc\minitoc
\renewcommand{\minitoc}{%
  \let\@orig@contentsline\contentsline
  \let\contentsline\@pcann@annotated@cl
  \@orig@minitoc
  \let\contentsline\@orig@contentsline
}

% Single annotation right after the heading (body display + aux registration).
\newcommand{\pcstart}[2]{%
  \ifpagecount
    \label{pcs:#1}%
    \expandafter\gdef\csname pct:#1\endcsname{#2}%
    \expandafter\gdef\csname pcid@\@currentHref\endcsname{#1}%
    \protected@write\@auxout{}{\string\pcannmark{#1}{#2}{\@currentHref}}%
    \noindent{%
      {\color{accentSkyBlue}\footnotesize\itshape
        $\triangleright$\enspace target:~#2\,pp}%
      \enspace
      {\color{accentOrange}\footnotesize\itshape
        $\triangleleft$\enspace actual:~\the\numexpr
          \getpagerefnumber{pce:#1}-\getpagerefnumber{pcs:#1}\relax\,pp}%
    }\par\vspace{-\parskip}%
  \fi
}
