% Chapter epigraphs — single source of truth for all chapter quotes.
% Each chapter uses \QuoteChN (quote text) and \AttribChN (attribution), with N in
% Roman numerals (\QuoteChI, \QuoteChII, ...). They are printed only when
% \showquotestrue is set in references/style/main.tex.
%
% \sourcelink{URL}{text} renders TEXT as an invisible hyperlink (no color change).
% If URL is empty, it renders TEXT as plain text.
\newcommand{\sourcelink}[2]{%
  \ifx&#1&%
    #2%
  \else%
    {\hypersetup{hidelinks}\href{#1}{#2}}%
  \fi%
}
