% A table wider than the text is scaled down to \linewidth. One wider than 2/3 of
% the text width is set to the full width by widening its column separation, so its
% font keeps the text size; tables whose separation cannot grow (@{} columns) are scaled.
\newsavebox{\fitwidthbox}
\newsavebox{\fitwidthprobe}
\newlength{\fitwidthslope}
\newcommand{\fitwidth}[1]{%
  \begingroup
  \sbox{\fitwidthbox}{#1}%
  \ifdim\wd\fitwidthbox>\linewidth
    \resizebox{\linewidth}{!}{\usebox{\fitwidthbox}}%
  \else\ifdim\wd\fitwidthbox>0.66\linewidth
    \begingroup\addtolength{\tabcolsep}{1pt}\global\sbox{\fitwidthprobe}{#1}\endgroup
    \setlength{\fitwidthslope}{\dimexpr\wd\fitwidthprobe-\wd\fitwidthbox\relax}%
    \ifdim\fitwidthslope>0.5pt
      \setlength{\tabcolsep}{\dimexpr\tabcolsep+(\linewidth-\wd\fitwidthbox)*65536/\number\fitwidthslope\relax}%
      #1%
    \else
      \resizebox{\linewidth}{!}{\usebox{\fitwidthbox}}%
    \fi
  \else
    \usebox{\fitwidthbox}%
  \fi\fi
  \endgroup}
