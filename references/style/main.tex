% convenience commands for highlights
\newcommand{\maincolor}{\color{accentBlack}}

\newcommand{\secondary}[1]{\textcolor{accentSkyBlue}{#1}}

% Chapter epigraphs (references/quotes/quotes.tex): \showquotestrue to print them
\newif\ifshowquotes
\showquotestrue % \showquotesfalse hides every epigraph
\newcommand{\chaptertitleblock}[3]{%
  \ifshowquotes\if\relax\detokenize{#1}\relax
  \else
    \begin{displayquote}
      \textit{#1} \\
      --- \textup{#2}
    \end{displayquote}
  \fi\fi
  \minitoc
  \vspace{1em}
  #3
  \par\medskip
}

  \textcolor{black}{#1}%
} % Render all glossary-linked text in black while preserving hyperlinks
